\begin{frame}{Core Idea}
  \begin{block}{Story premise}
    Lan disappears under unusual circumstances. Minh and Nam follow her final recording, concealed records, and traces inside an old hospital to uncover the truth behind the case.
  \end{block}

  \vspace{0.35cm}
  \begin{itemize}
    \item \textbf{Genre:} exploration, investigation, environmental puzzles, and survival horror.
    \item \textbf{Player goal:} find Lan, expose the hidden case, and survive the final confrontation.
    \item \textbf{Emotional arc:} everyday unease develops into tension, fear, and psychological climax.
  \end{itemize}
\end{frame}

\begin{frame}{Four Maps Form One Continuous Story}
  \begin{center}
    \colorbox{GameLight}{\parbox{0.92\linewidth}{\centering\large
      \textbf{Dormitory} \quad$\longrightarrow$\quad
      \textbf{Police Station} \quad$\longrightarrow$\quad
      \textbf{Old Hospital} \quad$\longrightarrow$\quad
      \textbf{Killer's House}\\[0.25cm]
      A disappearance becomes a concealed investigation, a horror survival journey, and finally a confrontation with the truth.}}
  \end{center}

  \vspace{0.35cm}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
    \begin{block}{Story rhythm}
      Everyday mystery \,$\rightarrow$\, sealed records \,$\rightarrow$\, a real threat \,$\rightarrow$\, psychological climax.
    \end{block}
    \column{.48\textwidth}
    \begin{block}{Gameplay rhythm}
      Learn the mechanics \,$\rightarrow$\, investigate \,$\rightarrow$\, hide and escape \,$\rightarrow$\, confront and resolve.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{The Player Experience Loop}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
    \begin{block}{What the player does}
      \begin{enumerate}
        \item Explore 3D spaces and observe unusual details.
        \item Interact with NPCs, objects, locked doors, and evidence.
        \item Collect recordings, documents, items, and clues.
        \item Solve conditions or avoid danger to unlock the next objective.
      \end{enumerate}
    \end{block}
    \column{.48\textwidth}
    \begin{alertblock}{How tension is created}
      Strange sounds, flickering lights, dark corridors, restricted areas, patrolling characters, and chase sequences make every clue feel risky.
    \end{alertblock}
  \end{columns}
\end{frame}

\section{Four Chapters}
\begin{frame}{Chapter 1: Dormitory -- Lan's Last Recording}
  \begin{columns}[T,onlytextwidth]
    \column{.50\textwidth}
    \begin{block}{Story}
      Minh discovers that Lan is missing. Her final recording contains an unfamiliar background sound. Minh and Nam search the dormitory for notes, objects, and traces that connect the disappearance to an old case.
    \end{block}
    \column{.46\textwidth}
    \begin{block}{Gameplay role}
      This tutorial chapter introduces movement, sprinting, stamina, interaction, item pickup, notes, NPC dialogue, and clue storage.
    \end{block}
  \end{columns}

  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Areas:} Minh's room, Nam's room, Lan's belongings, computer/common room, storage, corridors, lobby, and dormitory gate.
    \item \textbf{Threats:} strange noises, flickering lights, the feeling of being watched, and light jump scares.
    \item \textbf{Chapter outcome:} a case code or the name of an old hospital leads the player to the police station.
  \end{itemize}
\end{frame}

\begin{frame}{Chapter 2: Police Station -- The Sealed File}
  \begin{columns}[T,onlytextwidth]
    \column{.50\textwidth}
    \begin{block}{Story}
      Minh and Nam search for records about Lan, the old hospital, earlier disappearances, and the people involved. They discover that part of the information has been deleted or deliberately hidden.
    \end{block}
    \column{.46\textwidth}
    \begin{block}{Gameplay role}
      The player begins sneaking, finding keys or access cards, avoiding patrols, assembling fragmented documents, and entering restricted areas.
    \end{block}
  \end{columns}

  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Areas:} public lobby, offices, duty room, interrogation room, evidence room, archive, temporary holding area, and rear parking lot.
    \item \textbf{Main objective:} reach the archive, assemble the sealed files, and obtain the hospital address or floor plan.
    \item \textbf{Chapter outcome:} Lan investigated the hospital, and someone inside the station may be hiding the truth.
  \end{itemize}
\end{frame}

\begin{frame}{Chapter 3: Old Hospital -- The Heart of the Horror}
  \begin{columns}[T,onlytextwidth]
    \column{.50\textwidth}
    \begin{block}{Story}
      Minh and Nam enter the old hospital where the case began. They investigate signs of illegal activity, experiments, or concealed crimes and find evidence that Lan was taken there.
    \end{block}
    \column{.46\textwidth}
    \begin{block}{Gameplay role}
      This is the strongest horror chapter: restore power, open sealed areas, collect critical evidence, hide, and escape from the pursuer.
    \end{block}
  \end{columns}

  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Areas:} reception, inpatient corridors, wards, treatment rooms, doctors' offices, morgue, technical basement, and quarantine area.
    \item \textbf{Environmental pressure:} unstable audio, flickering power, shifting corridors, sudden locked doors, and chase sequences.
    \item \textbf{Chapter outcome:} a photograph, recording, or notebook reveals that the killer is still alive and exposes the final address.
  \end{itemize}
\end{frame}

\begin{frame}{Chapter 4: Killer's House -- The Final Truth}
  \begin{columns}[T,onlytextwidth]
    \column{.50\textwidth}
    \begin{block}{Story}
      Minh and Nam follow the address to a house that looks ordinary outside but is distorted within. It preserves the killer's obsession with the hospital and previous victims.
    \end{block}
    \column{.46\textwidth}
    \begin{block}{Gameplay role}
      The highest-tension chapter: avoid the killer's patrol, overcome traps, open the path to a hidden lower level, and complete the final escape or confrontation.
    \end{block}
  \end{columns}

  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Areas:} front yard, living room, kitchen, bedrooms, study, locked room, basement, and holding/evidence area.
    \item \textbf{Main objective:} identify the killer, rescue Lan or discover her fate, and survive the final sequence.
    \item \textbf{Possible endings:} rescue Lan; fail after the evidence is destroyed; or discover in a hidden ending that Nam knows more than expected.
  \end{itemize}
\end{frame}

\section{Why This Structure Works}
\begin{frame}{Four Maps Control Scope and Escalate Emotion}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
    \begin{block}{Easier to develop}
      \begin{itemize}
        \item Four clear large maps instead of many disconnected small locations.
        \item Complete the gameplay loop in the first map, then reuse it in later maps.
        \item Every chapter has a clear goal, identity, and ending condition.
      \end{itemize}
    \end{block}
    \column{.48\textwidth}
    \begin{block}{A natural horror-investigation rhythm}
      \begin{itemize}
        \item Dormitory: familiar mystery with limited direct danger.
        \item Police station: surveillance and concealed information.
        \item Hospital: intense horror and real danger.
        \item Killer's house: pressure, confrontation, and resolution.
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}
